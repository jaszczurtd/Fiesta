cmake_minimum_required(VERSION 3.22)
project(fiesta_modules LANGUAGES C CXX)
get_filename_component(FIESTA_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(FIESTA_HAL_ROOT "${FIESTA_ROOT}/src/JaszczurHAL" CACHE PATH "HAL checkout")
set(JHBASE_ROOT "${FIESTA_HAL_ROOT}/third_party/jhbase" CACHE PATH "Pinned jhbase source")
set(JHBASE_HOST ON CACHE BOOL "Host files and JSON" FORCE)
add_subdirectory("${JHBASE_ROOT}" jhbase)
include("${FIESTA_ROOT}/src/common/cmake/fiesta_shared_sources.cmake")
set(shared_sets "")
foreach(name IN LISTS FIESTA_SHARED_SOURCE_SETS)
    string(APPEND shared_sets "\"${name}\",\n")
endforeach()
file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/fiesta_shared_sets.h"
    CONTENT "static const char *const fm_shared_sets[] = {\n${shared_sets}};\n")
add_library(fiesta_modules_core STATIC registry.c render.c support.c shared.c)
target_include_directories(fiesta_modules_core PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}"
    PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
target_link_libraries(fiesta_modules_core PUBLIC jb::host)
add_executable(fiesta-modules main.c)
target_link_libraries(fiesta-modules PRIVATE fiesta_modules_core)
file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/tool-$<CONFIG>.txt"
    CONTENT "$<TARGET_FILE:fiesta-modules>")
foreach(target fiesta_modules_core fiesta-modules)
    target_compile_features(${target} PRIVATE c_std_11)
    set_target_properties(${target} PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED ON C_EXTENSIONS OFF)
    jb_warnings(${target})
    if(MSVC)
        target_link_options(${target} INTERFACE /INCREMENTAL:NO)
    endif()
endforeach()
target_compile_definitions(fiesta_modules_core PRIVATE
    FIESTA_CMAKE_COMMAND="${CMAKE_COMMAND}"
    FIESTA_SHARED_READER="${FIESTA_ROOT}/src/common/cmake/fiesta_shared_sources.cmake")

option(BUILD_TESTING "Build native registry tests" OFF)
include("${FIESTA_ROOT}/src/common/cmake/fiesta_host_analysis.cmake")
include(CTest)
if(BUILD_TESTING)
    add_executable(fiesta_modules_tests tests/test_modules.c)
    target_link_libraries(fiesta_modules_tests PRIVATE fiesta_modules_core)
    target_compile_features(fiesta_modules_tests PRIVATE c_std_11)
    set_target_properties(fiesta_modules_tests PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED ON C_EXTENSIONS OFF)
    jb_warnings(fiesta_modules_tests)
    include("${FIESTA_HAL_ROOT}/cmake/jh_test_memory_limit.cmake")
    jh_limit_test_memory(fiesta_modules_tests)
    add_test(NAME fiesta_modules COMMAND fiesta_modules_tests
        "${CMAKE_CURRENT_BINARY_DIR}/test-work" "$<TARGET_FILE:fiesta-modules>" "${FIESTA_ROOT}")
    set_tests_properties(fiesta_modules PROPERTIES TIMEOUT 120 LABELS memcheck)
    fiesta_add_host_analysis_targets("module registry" FIESTA_MODULES
        CLANG_TIDY_FILES "tools/modules/.*\\.c$")
endif()
